use SHA256 simples por agora) async def hash_password(password): encoder = js.TextEncoder.new() data = encoder.encode(password) digest = await js.crypto.subtle.digest("SHA-256", data) hash_array = js.Uint8Array.new(digest) return ''.join(f'{byte:02x}' for byte in hash_array) async def cadastrar_usuario(username, password): password_hash = await hash_password(password) try: cursor.execute('INSERT INTO usuarios (username, password_hash) VALUES (?, ?)', (username, password_hash)) conn.commit() return True except: return False async def login(username, password): cursor.execute('SELECT id, password_hash FROM usuarios WHERE username = ?', (username,)) result = cursor.fetchone() if result: input_hash = await hash_password(password) if result[1] == input_hash: return result[0] return None # Outras funções: adicionar_transacao, listar_transacoes, etc. (porte o resto do main.py aqui, adaptando para async se necessário para JS interop) # Exemplo simplificado para adicionar transacao def adicionar_transacao(usuario_id, tipo, valor, categoria, descricao): data = datetime.now().strftime('%Y-%m-%d %H:%M:%S') cursor.execute('INSERT INTO transacoes (usuario_id, tipo, valor, categoria, descricao, data) VALUES (?, ?, ?, ?, ?, ?)', (usuario_id, tipo, valor, categoria, descricao, data)) conn.commit() # Função para mostrar output def show_output(text): Element("output").write(text) # Eventos de botões (interop) login_btn = Element("login-btn") login_btn.element.onclick = lambda e: asyncio.create_task(handle_login()) cadastro_btn = Element("cadastro-btn") cadastro_btn.element.onclick = lambda e: asyncio.create_task(handle_cadastro()) async def handle_login(): username = Element("username").value password = Element("password").value usuario_id = await login(username, password) if usuario_id: show_output("Login bem-sucedido! ID: " + str(usuario_id)) Element("login-form").element.style.display = "none" Element("main-menu").element.style.display = "block" # Aqui, adicione listeners para outros botões (ex: add-receita.onclick = ...) else: show_output("Credenciais inválidas.") async def handle_cadastro(): username = Element("username").value password = Element("password").value if await cadastrar_usuario(username, password): show_output("Usuário cadastrado com sucesso!") else: show_output("Erro ao cadastrar (usuário já existe?).") # Porte mais funções e botões conforme necessário; para gráficos, use plt e display(fig, target="output")
