packages = ["sqlite3", "pandas", "matplotlib", "requests"] import sqlite3 import pandas as pd import matplotlib.pyplot as plt from datetime import datetime import asyncio from pyscript import Element, display from pyodide.ffi import create_proxy # Importante para proxies em event listeners # Banco em memória conn = sqlite3.connect(':memory:') cursor = conn.cursor() cursor.execute('''CREATE TABLE IF NOT EXISTS usuarios ( id INTEGER PRIMARY KEY AUTOINCREMENT, username TEXT UNIQUE NOT NULL, password_hash TEXT NOT NULL )''') cursor.execute('''CREATE TABLE IF NOT EXISTS transacoes ( id INTEGER PRIMARY KEY AUTOINCREMENT, usuario_id INTEGER NOT NULL, tipo TEXT NOT NULL, valor REAL NOT NULL, categoria TEXT NOT NULL, descricao TEXT, data TEXT NOT NULL )''') conn.commit() # Funções auxiliares async def hash_password(password): from js import TextEncoder, crypto, Uint8Array encoder = TextEncoder.new() data = encoder.encode(password) digest = await crypto.subtle.digest("SHA-256", data) hash_array = Uint8Array.new(digest) return ''.join(f'{byte:02x}' for byte in hash_array) async def cadastrar_usuario(username, password): password_hash = await hash_password(password) try: cursor.execute('INSERT INTO usuarios (username, password_hash) VALUES (?, ?)', (username, password_hash)) conn.commit() return True except: return False async def login(username, password): cursor.execute('SELECT id, password_hash FROM usuarios WHERE username = ?', (username,)) result = cursor.fetchone() if result: input_hash = await hash_password(password) if result[1] == input_hash: return result[0] return None def adicionar_transacao(usuario_id, tipo, valor, categoria, descricao): data = datetime.now().strftime('%Y-%m-%d %H:%M:%S') cursor.execute('INSERT INTO transacoes (usuario_id, tipo, valor, categoria, descricao, data) VALUES (?, ?, ?, ?, ?, ?)', (usuario_id, tipo, valor, categoria, descricao, data)) conn.commit() def show_output(text): Element("output").write(text) # Handlers (async onde necessário) async def handle_login(event): username = Element("username").value password = Element("password").value usuario_id = await login(username, password) if usuario_id: show_output(f"Login bem-sucedido! ID: {usuario_id}") Element("login-form").element.style.display = "none" Element("main-menu").element.style.display = "block" else: show_output("Credenciais inválidas.") async def handle_cadastro(event): username = Element("username").value password = Element("password").value if await cadastrar_usuario(username, password): show_output("Usuário cadastrado com sucesso!") else: show_output("Erro ao cadastrar (usuário já existe?).") def handle_relatorio(event): df = pd.read_sql_query("SELECT tipo, valor, categoria, data FROM transacoes", conn) if df.empty: show_output("Nenhuma transação registrada.") return fig, ax = plt.subplots() df.groupby("tipo")["valor"].sum().plot(kind="bar", ax=ax) ax.set_title("Resumo por tipo de transação") display(fig, target="output") # Conectar eventos usando create_proxy para evitar destruição de proxy login_proxy = create_proxy(handle_login) Element("login-btn").element.addEventListener("click", login_proxy) cadastro_proxy = create_proxy(handle_cadastro) Element("cadastro-btn").element.addEventListener("click", cadastro_proxy) relatorio_proxy = create_proxy(handle_relatorio) Element("ver-relatorio").element.addEventListener("click", relatorio_proxy) # Para limpar proxies quando não necessários (opcional, para boa prática) # Ex: Element("login-btn").element.removeEventListener("click", login_proxy) # login_proxy.destroy()
